Practical 1: Block Diagram Reduction

Standard Functions
clc; clear; close all;
s = tf('s');

disp('--- Individual Transfer Functions ---');
G1 = 1/(s+1)
G2 = 2/(s+2)

disp('--- Overall Transfer Function (Series) ---');
series(G1,G2)

disp('--- Overall Transfer Function (Parallel) ---');
parallel(G1,G2)

disp('--- Overall Transfer Function (Feedback) ---');
feedback(G1,G2)

Practical 2: Step Response

The Trick: 4x Paste fprintf
clc; clear; close all; 
w = 5; z = 0.5;
sys = tf(w^2, [1 2*z*w w^2]); % Fixed: added missing '*' operators
disp('Transfer Function:'); sys

[y,t] = step(sys); 
figure; plot(t,y,'k','LineWidth',3); grid on;
title('Unit Step Response'); xlabel('Time'); ylabel('Amplitude');

i = stepinfo(sys); 
disp('--- Unit Step Response Parameters ---');
fprintf('Rise Time = %.4f sec\n', i.RiseTime);
fprintf('Peak Time = %.4f sec\n', i.PeakTime);
fprintf('Settling Time = %.4f sec\n', i.SettlingTime);
fprintf('Overshoot = %.2f %%\n', i.Overshoot);

Practical 3: Steady State Errors

The Trick: The "No IF" Hack
clc; clear; close all; 
syms s; G_s = 10/(s*(s+2));
disp('Symbolic TF:'); pretty(G_s);

s = tf('s'); G = 10/(s*(s+2)); 
kp = dcgain(G); 
kv = dcgain(s*G);   % Fixed: added '*'
ka = dcgain(s^2*G); % Fixed: added '*'

fprintf('\nSTEADY STATE ANALYSIS\n\n');
fprintf('Position error constant = %g\n', kp);
fprintf('Velocity error constant = %g\n', kv);
fprintf('Acceleration error constant = %g\n\n', ka);
fprintf('Unit Step error = %g\n', 1/(1+kp));
fprintf('Ramp error = %g\n', 1/kv);
fprintf('Parabolic error = %g\n', 1/ka);

Practical 4: Parameter Variation

The Trick: The Loop Paste (K, Z, W)
clc; clear; close all; 
w = 5; z = 0.5;

% LOOP 1: K
figure; hold on; 
for K=[1 5 10]
    step(tf(K*w^2, [1 2*z*w K*w^2])); % Fixed operators
end
legend('1','5','10'); title('Varying Gain K');

% LOOP 2: ZETA
figure; hold on; 
for Z=[0.2 0.5 0.8]
    step(tf(w^2, [1 2*Z*w w^2]));     % Fixed operators
end
legend('0.2','0.5','0.8'); title('Varying Zeta (Z)');

% LOOP 3: Wn
figure; hold on; 
for W=[2 5 8]
    step(tf(W^2, [1 2*z*W W^2]));     % Fixed operators
end
legend('2','5','8'); title('Varying Wn (W)');

Practical 5: Root Locus Analysis

The Trick: The Clean Click (rlocfind)
clc; clear; close all; 
G = tf(1, [1 7 10 0]);
figure; rlocus(G); grid on; title('Root Locus');

disp('Click the plot window to select a point!');
[K, p] = rlocfind(G);
disp('Selected Gain K:'); disp(K); 
disp('Closed Loop Poles:'); disp(p);

T = feedback(K*G, 1); 
figure; step(T); grid on; title('Step Response with Selected Gain');
disp('Parameters:'); disp(stepinfo(T));

Practical 6: Lead Compensation

The Trick: 1-Line System Comparison
clc; clear; close all;
G = tf(10, [1 1 0]); 
Gc = 0.9 * tf([1 1], [1 3]);

T1 = feedback(G, 1);

Practical 8: State Space Conversions

The Trick: The Transformer Paste
clc; clear; close all; 
n = [1 3]; d = [1 5 6]; 
G = tf(n,d);

[A,B,C,D] = tf2ss(n,d); 
sys = ss(A,B,C,D);

[n2,d2] = ss2tf(A,B,C,D); 
G2 = tf(n2,d2);

disp('--- Original TF ---'); G
disp('--- State-Space Model ---'); sys
disp('--- Converted TF back ---'); G2

figure; step(G, 'r', G2, 'b--'); grid on; 
legend('Original', 'Converted'); title('Transformation Validation');
